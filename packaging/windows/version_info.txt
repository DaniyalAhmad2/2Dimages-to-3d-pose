# Windows version resource, embedded into Pose3D.exe by pose3d.spec.
#
# Without this the file's Properties dialog is blank and SmartScreen has no
# publisher string to show at all, which makes an unsigned download look worse
# than it needs to. Keep the version numbers in step with pyproject.toml.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 1, 0, 0),
    prodvers=(0, 1, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [StringStruct('CompanyName', 'Pose3D'),
         StringStruct('FileDescription', '3D pose reconstruction from two camera views'),
         StringStruct('FileVersion', '0.1.0.0'),
         StringStruct('InternalName', 'Pose3D'),
         StringStruct('OriginalFilename', 'Pose3D.exe'),
         StringStruct('ProductName', 'Pose3D'),
         StringStruct('ProductVersion', '0.1.0.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
